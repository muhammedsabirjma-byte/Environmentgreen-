    <script>
        const imageLoader = document.getElementById('imageLoader');
        const canvas = document.getElementById('imageCanvas');
        const ctx = canvas.getContext('2d');
        const downloadBtn = document.getElementById('downloadBtn');
        const canvasContainer = document.getElementById('canvasContainer');

        const frameImg = new Image();
        frameImg.src = 'frame.png'; 

        let userImg = new Image();

        imageLoader.addEventListener('change', handleImage, false);

        function handleImage(e) {
            const reader = new FileReader();
            reader.onload = function(event) {
                userImg.onload = function() {
                    canvas.width = frameImg.width;
                    canvas.height = frameImg.height;

                    // ഫ്രെയിമിലെ പച്ച ചതുരത്തിന്റെ ഏകദേശ സ്ഥാനം ക്രമീകരിക്കുന്നത് (നിങ്ങളുടെ ചിത്രത്തിന് അനുസരിച്ച് മാറ്റാം)
                    let targetX = canvas.width * 0.05;   
                    let targetY = canvas.height * 0.61;  
                    let targetWidth = canvas.width * 0.42; 
                    let targetHeight = canvas.height * 0.25; 

                    ctx.drawImage(userImg, targetX, targetY, targetWidth, targetHeight);
                    ctx.drawImage(frameImg, 0, 0);

                    canvasContainer.style.display = 'block';
                    downloadBtn.style.display = 'inline-block';
                }
                userImg.src = event.target.result;
            }
            reader.readAsDataURL(e.target.files[0]);
        }

        downloadBtn.addEventListener('click', function() {
            const link = document.createElement('a');
            link.download = 'environment_day_poster.png';
            link.href = canvas.toDataURL();
            link.click();
        });
    </script>
